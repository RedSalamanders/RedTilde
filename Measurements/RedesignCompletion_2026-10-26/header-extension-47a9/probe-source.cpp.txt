#include <DxUi/DxUi.h>
#include <DxUi/RichDocument.h>

#include <concepts>
#include <cstdio>
#include <type_traits>

template <typename T>
concept HasReadOnlySourceSnapshotSynchronization =
    requires(T& editor, const DxUi::RichText::SourceSnapshot& source, DxUi::RichText::Selection selection)
{
    { editor.SynchronizeReadOnlySourceSnapshot(source, selection) } -> std::same_as<bool>;
};

#if DXUI_PROBE_EXPECT_READONLY_SYNC
using ReadOnlySourceSnapshotSynchronization = bool (DxUi::RichTextEditor::*)(
    const DxUi::RichText::SourceSnapshot&, DxUi::RichText::Selection);
static_assert(HasReadOnlySourceSnapshotSynchronization<DxUi::RichTextEditor>);
static_assert(std::is_same_v<decltype(&DxUi::RichTextEditor::SynchronizeReadOnlySourceSnapshot),
    ReadOnlySourceSnapshotSynchronization>);
#else
static_assert(! HasReadOnlySourceSnapshotSynchronization<DxUi::RichTextEditor>);
#endif

static_assert(sizeof(DxUi::RichTextEditor) > 0);
static_assert(sizeof(DxUi::RichText::Document) > 0);
static_assert(sizeof(DxUi::RichText::SourceSnapshot) > 0);
static_assert(alignof(DxUi::RichTextEditor) > 0);
static_assert(alignof(DxUi::RichText::Document) > 0);
static_assert(alignof(DxUi::RichText::SourceSnapshot) > 0);

#if defined(_ITERATOR_DEBUG_LEVEL)
constexpr int iteratorDebugLevel = _ITERATOR_DEBUG_LEVEL;
#else
constexpr int iteratorDebugLevel = -1;
#endif

int main()
{
    std::printf(
        "{\"sizeofRichTextEditor\":%zu,\"alignofRichTextEditor\":%zu,"
        "\"sizeofRichTextDocument\":%zu,\"alignofRichTextDocument\":%zu,"
        "\"sizeofSourceSnapshot\":%zu,\"alignofSourceSnapshot\":%zu,"
        "\"controlIsPolymorphic\":%s,\"richTextEditorIsPolymorphic\":%s,"
        "\"hasReadOnlySourceSynchronization\":%s,"
        "\"msvcVersion\":%d,\"msvcLang\":%ld,\"iteratorDebugLevel\":%d}\n",
        sizeof(DxUi::RichTextEditor),
        alignof(DxUi::RichTextEditor),
        sizeof(DxUi::RichText::Document),
        alignof(DxUi::RichText::Document),
        sizeof(DxUi::RichText::SourceSnapshot),
        alignof(DxUi::RichText::SourceSnapshot),
        std::is_polymorphic_v<DxUi::Control> ? "true" : "false",
        std::is_polymorphic_v<DxUi::RichTextEditor> ? "true" : "false",
        HasReadOnlySourceSnapshotSynchronization<DxUi::RichTextEditor> ? "true" : "false",
#if defined(_MSC_VER)
        _MSC_VER,
#else
        0,
#endif
        static_cast<long>(_MSVC_LANG),
        iteratorDebugLevel);
    return 0;
}
